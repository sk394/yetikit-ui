## Usage

```tsx
import { AnimatedNumber } from '@/components/ui/animated-number';
```

```tsx
<AnimatedNumber value={128430} className="text-4xl font-semibold" />
```

Pass a `format` function for currency or units:

```tsx
<AnimatedNumber
  value={4521.5}
  format={(n) => new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(n)}
/>
```

<TypeTable
